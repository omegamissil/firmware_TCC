Para adicionar uma indicação visual clara dos estados do sistema (Normal, Alerta e Crítico), a melhor abordagem com o ESP32 é utilizar um **LED RGB com 4 pinos (tipo Catodo Comum)** ou um **LED Endereçável WS2812B (NeoPixel)**. O LED RGB comum de 4 pinos é a opção mais direta e econômica, utilizando resistores limitadores de corrente e três pinos digitais com modulação PWM para controlar a intensidade e mesclar as cores.

Abaixo estão os detalhes das ligações físicas e o código atualizado do firmware integrando a sinalização luminosa.

---

### 1. Ligações Físicas do LED RGB (Catodo Comum)

Em um LED RGB de catodo comum, o pino mais longo é o **GND (Terra)**, e os outros três controlam as cores Vermelho, Verde e Azul.

| Pino do LED RGB | Pino Correspondente no ESP32 | Componente Intermediário | Função / Descrição |
| --- | --- | --- | --- |
| **Pino Longo (Catodo / GND)** | Pino `GND` | Nenhum | Terra comum do LED |
| **Pino Vermelho (R)** | Pino `GPIO 18` | Resistor de $220\,\Omega$ em série | Acende na cor Vermelha (Estado Crítico) |
| **Pino Verde (G)** | Pino `GPIO 19` | Resistor de $220\,\Omega$ em série | Acende na cor Verde (Estado Normal) |
| **Pino Azul (B)** | Pino `GPIO 23` | Resistor de $220\,\Omega$ em série | Acende na cor Azul (Estado de Alerta) |

> **Nota de Segurança:** Sempre utilize resistores (entre $220\,\Omega$ e $330\,\Omega$) em série com os pinos de cor do LED RGB para limitar a corrente e proteger os pinos GPIO do ESP32.

---

### 2. Mapeamento de Cores para os Estados do Sistema

* **Estado Normal ($T \le 27^\circ\text{C}$):** **Cor Verde** (indicando operação estável).
* **Estado de Alerta ($27^\circ\text{C} < T \le 32^\circ\text{C}$):** **Cor Azul** (indicando ventilação auxiliar ativa).
* **Estado Crítico ($T > 32^\circ\text{C}$):** **Cor Vermelha** (indicando alarme sonoro e emergência).

---

unsigned long ultimaLeitura = 0;

// Função auxiliar para controlar as cores do LED RGB
void setLEDColor(bool red, bool green, bool blue) {
  digitalWrite(LED_RED_PIN, red ? HIGH : LOW);
  digitalWrite(LED_GREEN_PIN, green ? HIGH : LOW);
  digitalWrite(LED_BLUE_PIN, blue ? HIGH : LOW);
}

void setup() {
  Serial.begin(115200);
  
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_RED_PIN, OUTPUT);
  pinMode(LED_GREEN_PIN, OUTPUT);
  pinMode(LED_BLUE_PIN, OUTPUT);

  digitalWrite(RELAY_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);
  setLEDColor(false, false, false);

  dht.begin();
  Wire.begin(21, 22);
  lcd.init();
  lcd.backlight();
  
  lcd.setCursor(0, 0);
  lcd.print("Iniciando IoT...");
  lcd.setCursor(0, 1);
  lcd.print("Server Rack Mon.");

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  lcd.clear();
  lcd.print("Wi-Fi Conectado!");
  delay(1500);

  espClient.setCACert(root_ca);
  client.setServer(mqtt_server, mqtt_port);
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Tentando conexão MQTTS...");
    if (client.connect("ESP32_ServerRoom_Monitor", token, "")) {
      Serial.println("[CONECTADO VIA TLS 1.2]");
    } else {
      delay(5000);
    }
  }
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  unsigned long tempoAtual = millis();
  
  if (tempoAtual - ultimaLeitura >= intervaloLeitura) {
    ultimaLeitura = tempoAtual;

    float umidade = dht.readHumidity();
    float temperatura = dht.readTemperature();

    if (isnan(umidade) || isnan(temperatura)) {
      Serial.println("Erro ao ler dados do sensor DHT22!");
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Erro Sensor DHT!");
      setLEDColor(true, true, true); // Branco indicando erro de leitura
      return;
    }

    Serial.printf("Temperatura: %.2f C | Umidade: %.2f %%\n", temperatura, umidade);

    // Atualização do LCD
    lcd.setCursor(0, 0);
    lcd.print("Temp: ");
    lcd.print(temperatura, 1);
    lcd.print((char)223);
    lcd.print("C   ");

    lcd.setCursor(0, 1);
    lcd.print("Umid: ");
    lcd.print(umidade, 1);
    lcd.print("%    ");

    // ==========================================
    // MÁQUINA DE ESTADOS, ATUADORES E LED RGB
    // ==========================================
    if (temperatura > TEMP_LIMITE_CRITICO) {
      digitalWrite(RELAY_PIN, HIGH);
      digitalWrite(BUZZER_PIN, HIGH);
      setLEDColor(true, false, false); // Vermelho (Crítico)
      Serial.println("[CRÍTICO] Temperatura > 32°C. Alarme e LED Vermelho ativos.");
    } 
    else if (temperatura > TEMP_LIMITE_ALERTA) {
      digitalWrite(RELAY_PIN, HIGH);
      digitalWrite(BUZZER_PIN, LOW);
      setLEDColor(false, false, true); // Azul (Alerta)
      Serial.println("[ALERTA] Temperatura > 27°C. Ventilação e LED Azul ativos.");
    } 
    else {
      digitalWrite(RELAY_PIN, LOW);
      digitalWrite(BUZZER_PIN, LOW);
      setLEDColor(false, true, false); // Verde (Normal)
    }

    // Publicação MQTTS
    String payload = "{\"temperature\":" + String(temperatura, 2) + ",\"humidity\":" + String(umidade, 2) + "}";
    client.publish("v1/devices/me/telemetry", payload.c_str());
  }
}
